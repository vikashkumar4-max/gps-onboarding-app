// ⚠️ यहाँ अपना Deploy किया हुआ Apps Script Web App URL पेस्ट करें const API_URL = "https://script.google.com/macros/s/AKfycbyNb329KNwYVoC-f0ceBIWg_oBFLGJlvE54kBFtEmCnJwcnrHsm6DMQ2aipIxI83i7W_g/exec"; let vehicleData = []; function loadData() { fetch(API_URL, { method: "GET", redirect: "follow" }) .then(res => res.json()) .then(data => { vehicleData = data; document.getElementById('loader').style.display = 'none'; document.getElementById('noDataMsg').style.display = 'block'; }) .catch(err => { document.getElementById('loader').innerHTML = "डेटा लोड करने में समस्या आई!"; }); } // 1. स्मार्ट क्लीनर: किसी भी फ़ॉर्मेट (स्पेस, डेश, डॉट्स) को हटाकर केवल प्योर टेक्स्ट/नंबर निकालता है function sanitize(str) { if (!str) return ''; return str.toString().replace(/[^a-zA-Z0-9]/g, '').toUpperCase(); } // 2. ऑटो-डिटेक्ट गाड़ी नंबर (किसी भी कॉलम हेडर नाम से फ़ेच करने के लिए) function getVehicleNo(row) { if (!row) return ''; const keys = Object.keys(row); // पहले सम्भावित कॉलम नाम चेक करें for (let k of keys) { let keyLower = k.toLowerCase(); if (keyLower.includes('veh') || keyLower.includes('reg') || keyLower.includes('number') || keyLower.includes('no')) { if (row[k]) return row[k].toString(); } } // अगर हेडर मैच न हो तो पहला कॉलम उठाएं return row[keys[0]] ? row[keys[0]].toString() : ''; } function getRemarkVal(row) { if (!row) return 'NO INFO'; const keys = Object.keys(row); for (let k of keys) { let keyLower = k.toLowerCase(); if (keyLower.includes('remark') || keyLower.includes('status')) { if (row[k]) return row[k].toString(); } } return 'NO INFO'; } // 3. AUTO-FETCH SEARCH FUNCTION function searchVehicle() { const rawQuery = document.getElementById('searchVehicle').value; const cleanQuery = sanitize(rawQuery); const container = document.getElementById('resultContainer'); const noDataMsg = document.getElementById('noDataMsg'); container.innerHTML = ''; // केवल 2 अक्षर टाइप करते ही सर्च शुरू हो जाएगा if (cleanQuery.length < 2) { noDataMsg.innerText = "गाड़ी नंबर सर्च करें..."; noDataMsg.style.display = 'block'; return; } // किसी भी फ़ॉर्मेट में डेटा मौजूद हो, यह उसे स्मार्टली फ़िल्टर करेगा const filtered = vehicleData.filter(row => { let vehClean = sanitize(getVehicleNo(row)); return vehClean.includes(cleanQuery); }); if (filtered.length === 0) { noDataMsg.innerText = "कोई गाड़ी नहीं मिली!"; noDataMsg.style.display = 'block'; return; } noDataMsg.style.display = 'none'; filtered.forEach((row, index) => { const vehNo = getVehicleNo(row) || 'N/A'; const remark = getRemarkVal(row); const isNoInfo = sanitize(remark).includes('NOINFO') || remark.trim() === ''; const card = document.createElement('div'); card.className = 'card'; card.innerHTML = `

${vehNo}

${remark}
${isNoInfo ? `
ALERT: गाड़ी LIVE करवाना आवश्यक है!
` : ''}
`; container.appendChild(card); }); } function submitGroundUpdate(vehNo, index) { const status = document.getElementById(`groundStatus-${index}`).value; const remark = document.getElementById(`groundRemark-${index}`).value; alert(`गाड़ी ${vehNo} का स्टेटस "${status}" अपडेट किया जा रहा है...`); fetch(API_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ vehicleNo: vehNo, status: status, remark: remark }) }) .then(() => { alert("स्टेटस सफलतापूर्वक अपडेट हो गया!"); loadData(); }) .catch(err => alert("अपडेट में त्रुटि हुई: " + err)); } loadData();
